\title{Two Formulations for Transmission-Network Reduction}
\author{Omid Mokhtari}

\begin{document}
\maketitle

\begin{abstract}
We formulate transmission-network reduction as a mixed-integer program that keeps a subset of buses and lines while bounding the resulting change in DC line flows. A first formulation assigns each bus to a retained bus through a binary assignment matrix; the resulting constraints are bilinear in the assignment and the retained-bus angles. A second, edge-based formulation avoids this by working directly with line-status variables, with the clustering recovered after solving as the connected components of the retained-line subgraph. The two formulations coincide under connected clustering, and we give a data-only bound for the auxiliary variable the edge-based formulation requires.
\end{abstract}

\begin{IEEEkeywords}
Transmission-network reduction, DC power flow, mixed-integer linear programming.
\end{IEEEkeywords}

\section{Assignment-Based Formulation}

Consider a connected transmission network with $N$ buses and $L$ lines. The data are
\begin{align}
 p&\in\R^N, &
 \hat\theta&\in\R^N, &
 D^x&\in\R^{L\times L}, &
 E&\in\R^{N\times L},
\end{align}
where $p$ is the nodal injection vector, $\hat\theta$ the original voltage-angle vector, $D^x$ the diagonal line-susceptance matrix, and $E$ the node--line incidence matrix, satisfying the original DC power flow
\begin{align}
 p&=ED^xE^{\mathsf T}\hat\theta, &
 \hat f&=D^xE^{\mathsf T}\hat\theta.
 \label{eq:original_flows}
\end{align}

A binary matrix $A\in\{0,1\}^{N\times N}$ defines the reduction: $A_{ij}=1$ iff bus $j$ is assigned to retained bus $i$. Each bus is assigned once, and only a retained bus can receive assignments:
\begin{align}
 \sum_{i=1}^{N}A_{ij}&=1, &&\forall j,
 \label{eq:assign_once}\\
 A_{ij}&\le A_{ii}, &&\forall i,j,
 \label{eq:only_retained}
\end{align}
with $A\in\mathcal A_{\mathrm{conn}}$ restricting feasible $A$ to those under which every cluster $\mathcal C_i:=\{j:A_{ij}=1\}$ induces a connected subgraph (e.g., a single-commodity spanning-forest formulation), and objective $\min\tr(A)$.

Line $l$ with endpoints $u_l,v_l$ is \emph{internal} if both endpoints share a retained bus, \emph{external} otherwise; internal lines disappear after contraction. With candidate retained-bus angles $\theta\in\R^N$, the angle assigned to every original bus is $A^{\mathsf T}\theta$, so the reduced flow and assigned balance are
\begin{align}
 f&=D^xE^{\mathsf T}A^{\mathsf T}\theta,
 \label{eq:original_flow}\\
 Ap&=AEf=AED^xE^{\mathsf T}A^{\mathsf T}\theta.
 \label{eq:original_balance}
\end{align}

External lines carry an accuracy and a physical-limit constraint relative to the original flow $\hat f_l$, with tolerance $\epsilon_l^L$:
\begin{align}
 |f_l-\hat f_l|&\le\epsilon_l^L,
 &&\forall l\text{ external},
 \label{eq:err_ext}\\
 \underline f_l\le f_l&\le\overline f_l,
 &&\forall l\text{ external}.
 \label{eq:lim_ext}
\end{align}
An absorbed line has equal endpoint angles, so \eqref{eq:original_flow} gives $f_l=0$; internal lines are exempt from \eqref{eq:err_ext}--\eqref{eq:lim_ext}, since contraction removes the line and the power it carried reappears as internal redistribution (Section~\ref{sec:edge}). Lines already at their limit,
\begin{equation}
 \mathcal L^+:=\{l:\hat f_l=\overline f_l\},\qquad
 \mathcal L^-:=\{l:\hat f_l=\underline f_l\},
\end{equation}
are kept external so that an existing congestion cannot disappear through aggregation:
\begin{align}
 f_l\ge\hat f_l,\ \forall l\in\mathcal L^+,\qquad
 f_l\le\hat f_l,\ \forall l\in\mathcal L^-.
 \label{eq:Lpm}
\end{align}

This model is exact, but \eqref{eq:original_flow} and \eqref{eq:original_balance} both contain products of the binary assignment $A$ with the continuous angles $\theta$, which is what Section~\ref{sec:edge} removes.

\section{Edge-Based Formulation}
\label{sec:edge}

\subsection{Line-Status Variables and Physical Flow}

Introduce $c_l\in\{0,1\}$, $c_l=1$ iff $u_l,v_l$ share a cluster, together with a lifted angle vector $\vartheta\in\R^N$ and
\begin{equation}
 f=D^xE^{\mathsf T}\vartheta,
 \qquad
 f_l=D^x_{ll}(\vartheta_{u_l}-\vartheta_{v_l}).
 \label{eq:physical_flow}
\end{equation}
This is the DC line-flow equation, not a mere definition of $f$: eliminating $\vartheta$ confines $f$ to the range of $D^xE^{\mathsf T}$, an $(N-1)$-dimensional subspace, which in a meshed network imposes one linear restriction (Kirchhoff's voltage law) per independent cycle. Contracting a cluster deletes its internal lines from that restriction, the sense in which an absorbed line acts as an infinite-susceptance connection.

With critical lines forced external, $c_l=0$ for $l\in\mathcal L^+\cup\mathcal L^-$, and the window
\begin{align}
 \underline\phi_l&:=\max\{\underline f_l,\hat f_l-\epsilon_l^L\}\ (\text{clipped to }\hat f_l\text{ on }\mathcal L^+),\\
 \overline\phi_l&:=\min\{\overline f_l,\hat f_l+\epsilon_l^L\}\ (\text{clipped to }\hat f_l\text{ on }\mathcal L^-),
 \label{eq:phi}
\end{align}
impose
\begin{equation}
 \underline\phi_l(1-c_l)\le f_l\le\overline\phi_l(1-c_l)
 \qquad \forall l,
 \label{eq:f_onoff}
\end{equation}
which gives $\underline\phi_l\le f_l\le\overline\phi_l$ when $c_l=0$ (external) and $f_l=0$ when $c_l=1$ (internal). A line in $\mathcal L^+\cup\mathcal L^-$ has $\underline\phi_l=\overline\phi_l=\hat f_l$, so it is reproduced exactly regardless of $\epsilon_l^L$. Since $f_l=0$ on an internal line forces $\vartheta_{u_l}=\vartheta_{v_l}$, connectivity of the clusters makes $\vartheta$ constant on each one, so $\vartheta=A^{\mathsf T}\theta$ for some $\theta$ -- obtained automatically, with no binary--continuous product.

\subsection{Internal Redistribution}

The flow $f$ balances the reduced network only at cluster level, not at every eliminated bus. Introduce $\gint\in\R^L$, an auxiliary transfer used only inside clusters -- not a physical DC flow -- with
\begin{equation}
 p=E\bigl(f+\gint\bigr),
 \qquad
 -G_l\,c_l\le \gint_l\le G_l\,c_l\ \ \forall l,
 \label{eq:split_balance}
\end{equation}
so $f$ carries power between clusters and $\gint$ redistributes injections inside them, on internal lines only, with $G_l$ a bound computed from the input data alone (Section~\ref{sec:tight}) so that it holds regardless of which clustering the model selects. The bound $\gint_l=0$ on external lines cannot be dropped: since $\mathbf 1^{\mathsf T}E=0$ and $\mathbf 1^{\mathsf T}p=0$, $p-Ef$ is zero-sum for every $f$, so an unbounded $\gint$ would make \eqref{eq:split_balance} solvable for any $f$ and let the whole network contract to one bus. A single reference angle $\vartheta_{j_0}=0$ fixes the remaining gauge freedom.

\subsection{Complete Model}

The reformulation removes the $O(N^2)$ assignment--angle products, replacing them with line-level binaries and a bounded internal transfer:
\begin{subequations}
\label{eq:edge_model}
\begin{align}
\max_{c,\vartheta,f,\gint}\quad
&\sum_{l=1}^{L} c_l
\\
\mathrm{s.t.}\quad
&c_l\in\{0,1\},\ \ c_l=0\ \forall l\in\mathcal L^+\cup\mathcal L^-,
\\
&f=D^xE^{\mathsf T}\vartheta,
\\
&\underline{\phi}_l(1-c_l)\le f_l\le\overline{\phi}_l(1-c_l),
&&\forall l,
\\
&p=E(f+\gint),
\\
&-G_l c_l\le \gint_l\le G_l c_l,
&&\forall l,
\\
&\vartheta_{j_0}=0.
\end{align}
\end{subequations}
The clustering is recovered after solving as the connected components of $\{l:c_l=1\}$ -- connected by construction, at no extra constraint, so the assignment matrix and $\mathcal A_{\mathrm{conn}}$ of Section~I are not needed at all. The objective maximizes internal lines rather than retained buses: at optimality, every non-protected line whose endpoints are already connected internally is also internal, since its flow is already zero and switching it on only improves the objective. Two clusterings with the same bus count can therefore score differently. Susceptances act only through \eqref{eq:physical_flow} -- retained lines keep their original values and no equivalent admittances are created, unlike Ward-type reduction.

\subsection{The Internal-Transfer Bound}
\label{sec:tight}

$G_l$ must hold for every clustering the model may choose, so it cannot depend on one; it is fixed from the input data before solving.

Let $\widehat f$ be the original DC flow, $e^0:=p-E\widehat f$ the (normally zero) original balance mismatch, and $d:=\widehat f-f$ the flow deviation, so $p-Ef=e^0+Ed$. For a line that may become internal, $f_l\in\{0\}\cup[\underline\phi_l,\overline\phi_l]$, giving a valid interval $[\underline d_l,\overline d_l]$ for $d_l$ from $\{\widehat f_l,\ \widehat f_l-\overline\phi_l,\ \widehat f_l-\underline\phi_l\}$ (for a line fixed external, only the last two). Propagating this interval through $e^0+Ed$ gives per-bus residual bounds $R_i^+,R_i^-\ge0$, whose sum, taken over both directions of the network, bounds every used internal edge:
\begin{equation}
 |\gint_l|\le G_{\mathrm{res}}:=\min\Bigl\{\textstyle\sum_iR_i^+,\ \sum_iR_i^-\Bigr\}.
 \label{eq:G_residual}
\end{equation}
A simpler alternative follows from $\Delta_l:=\max\{|\underline d_l|,|\overline d_l|\}$ and $\|e^0+Ed\|_1\le\|e^0\|_1+2\sum_l\Delta_l$:
\begin{equation}
 G_{\Delta}:=\tfrac12\|e^0\|_1+\sum_{l=1}^{L}\Delta_l.
 \label{eq:G_delta}
\end{equation}
The bound applied per line is the minimum of the two,
\begin{equation}
 G_l=\min\bigl\{G_{\mathrm{res}},\ G_{\Delta}\bigr\},
 \label{eq:G_recommended}
\end{equation}
which is itself valid as a minimum of independently valid bounds, and depends only on the admissible \emph{deviation} from the original flow rather than on the full injection and line-flow magnitudes.
